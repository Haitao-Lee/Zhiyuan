# Zhiyuan UI Loading Mechanism - Migration Guide
# ======================================================
#
# This document explains how to migrate from .ui file loading
# to compiled .pyd module usage in Slicer plugins.
#
# IMPORTANT: After Cython compilation, .ui files are converted to .py,
# and then compiled into .pyd. The original loadUI() calls need adjustment.
#

## Scenario 1: Standard Slicer Plugin UI Loading

### Before (Original .ui approach):
```python
class BrachyPlanWidget(ScriptedLoadableModuleWidget):
    def setup(self):
        # Load .ui file at runtime
        ScriptedLoadableModuleWidget.setup(self)
        
        uiFile = os.path.join(
            os.path.dirname(__file__).replace('\\', '/'),
            'Resources', 'UI',
            'BrachyPlan.ui'
        )
        self.ui = qt.QWidget()
        slicer.util.loadUI(uiFile, self.ui)
```

### After (Compiled .pyd approach):
```python
class BrachyPlanWidget(ScriptedLoadableModuleWidget):
    def setup(self):
        ScriptedLoadableModuleWidget.setup(self)
        
        # Option A: Import compiled UI class directly (RECOMMENDED)
        # The .ui was compiled to BrachyPlan_ui.py by smart_builder.py
        try:
            import BrachyPlan_ui as compiled_ui
            
            # Create instance of the compiled UI class
            self.ui = compiled_ui.Ui_BrachyPlan()
            
            # Create a QWidget to hold the UI
            self.widget = qt.QWidget()
            self.ui.setupUi(self.widget)  # Call generated setupUi method
            
            # Access widgets using the compiled object
            self.ui.planButton = self.ui.planButton  # Direct attribute access
            self.ui.imageSelector = self.ui.imageSelector
            
            # Set up connections as usual
            self.ui.planButton.connect('clicked(bool)', self.onPlanButton)
            
        except ImportError:
            # Fallback: Load .ui directly if compilation failed
            log.warning("Compiled UI not found, falling back to .ui loading")
            uiFile = os.path.join(
                os.path.dirname(__file__).replace('\\', '/'),
                'Resources', 'UI',
                'BrachyPlan.ui'
            )
            self.ui = qt.QWidget()
            slicer.util.loadUI(uiFile, self.ui)
            self.widget = self.ui
```

## Scenario 2: Using setupUI Pattern (PyQt/PySide style)

### When .ui is compiled with pyuic --execute flag:

The generated .py file contains:
- `class Ui_BrachyPlan(object)` - Main UI class
- `setupUi()` method - Sets up all widgets
- `retranslateUi()` method - For i18n support

### Usage Example:

```python
# In your main module (BrachyPlan.py)

# At the top of the file, after imports:
try:
    # Import the compiled UI module (generated by smart_builder.py)
    from BrachyPlan_ui import Ui_BrachyPlan
    HAS_COMPILED_UI = True
except ImportError:
    HAS_COMPILED_UI = False


class BrachyPlanWidget(ScriptedLoadableModuleWidget):
    """Main widget for BrachyPlan module."""
    
    def __init__(self, parent=None):
        ScriptedLoadableModuleWidget.__init__(self, parent)
        self._compiled_ui_instance = None
        
    def setup(self):
        ScriptedLoadableModuleWidget.setup(self)
        
        if HAS_COMPILED_UI:
            self._setup_with_compiled_ui()
        else:
            self._setup_with_ui_file()
    
    def _setup_with_compiled_ui(self):
        """Setup UI using pre-compiled .pyd module."""
        
        # Create the compiled UI instance
        self._compiled_ui_instance = Ui_BrachyPlan()
        
        # Create widget container
        self.widget = qt.QWidget()
        
        # Apply UI layout
        self._compiled_ui_instance.setupUi(self.widget)
        
        # Store references to commonly used widgets for convenience
        self.ui = self._compiled_ui_instance  # Maintain backward compatibility
        
        # Map UI elements to attributes (same names as in .ui file)
        self.imageSelector = self._compiled_ui_instance.imageSelector
        self.CTVImageSelector = self._compiled_ui_instance.CTVImageSelector
        self.OARImageSelector = self._compiled_ui_instance.OARImageSelector
        self.planButton = self._compiled_ui_instance.planButton
        self.segmentationButton = self._compiled_ui_instance.segmentationButton
        self.doseVisualizationCollapsibleButton = (
            self._compiled_ui_instance.doseVisualizationCollapsibleButton
        )
        
        # Set up collapsible buttons
        self.ui.collapsibleButtons = {}
        for btn_name in [
            'segmentationCollapsibleButton',
            'planningParamsCollapsibleButton',
            'doseVisualizationCollapsibleButton',
            'refDirecCollapsibleButton',
        ]:
            if hasattr(self._compiled_ui_instance, btn_name):
                btn = getattr(self._compiled_ui_instance, btn_name)
                if btn is not None:
                    self.ui.collapsibleButtons[btn_name] = btn
        
        # Add widget to layout
        self.layout.addWidget(self.widget)
    
    def _setup_with_ui_file(self):
        """Fallback: Load .ui file directly."""
        uiFile = os.path.join(
            os.path.dirname(os.path.abspath(__file__)).replace('\\', '/'),
            'Resources', 'UI',
            'BrachyPlan.ui'
        )
        if not os.path.exists(uiFile):
            raise FileNotFoundError(f"UI file not found: {uiFile}")
        
        self.widget = qt.QWidget()
        slicer.util.loadUI(uiFile, self.widget)
        self.ui = self.widget
    
    def cleanup(self):
        """Clean up when closing."""
        ScriptedLoadableModuleWidget.cleanup(self)
        self._compiled_ui_instance = None
```

## Scenario 3: Multiple UI Files

If you have multiple .ui files (e.g., Settings.ui, PlanningParameters.ui):

```python
# All .ui files are compiled to separate .py/.pyd modules
try:
    from BrachyPlan_ui import Ui_BrachyPlan          # Main UI
    from PlanningParameters_ui import Ui_PlanningParameters  # Parameters UI
    from Settings_ui import Ui_Settings                   # Settings UI
    COMPILED_UIS_AVAILABLE = True
except ImportError:
    COMPILED_UIS_AVAILABLE = False


class BrachyPlanWidget(ScriptedLoadableModuleWidget):
    def setup(self):
        ScriptedLoadableModuleWidget.setup(self)
        
        if COMPILED_UIS_AVAILABLE:
            # Initialize each UI component
            self.main_ui = Ui_BrachyPlan()
            self.params_ui = Ui_PlanningParameters()
            self.settings_ui = Ui_Settings()
            
            # Create containers
            self.main_widget = qt.QWidget()
            self.params_widget = qt.QWidget()
            self.settings_widget = qt.QWidget()
            
            # Setup each
            self.main_ui.setupUi(self.main_widget)
            self.params_ui.setupUi(self.params_widget)
            self.settings_ui.setupUi(self.settings_widget)
            
            # Combine into main layout
            self.layout.addWidget(self.main_widget)
            self.layout.addWidget(self.params_widget)
            self.layout.addWidget(self.settings_widget)
```

## Key Points Summary

| Aspect | Original (.ui) | Compiled (.pyd) |
|--------|---------------|----------------|
| **Loading** | `slicer.util.loadUI()` | `Ui_Class().setupUi(widget)` |
| **Access** | `self.ui.property` | `self.compiled_ui.property` |
| **Performance** | Parse .ui at runtime | Pre-compiled, faster |
| **Protection** | Source visible | Binary (.pyd), protected |
| **Debugging** | Easy (edit .ui) | Need recompilation |
